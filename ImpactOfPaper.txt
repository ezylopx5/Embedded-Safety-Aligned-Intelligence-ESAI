### **2. Softmin Reference Distribution (Equation 6)**

```python
def compute_softmin_reference(self, E_preds: torch.Tensor):
    # Compute risk: ||E_pred||_2
    risks = torch.norm(E_preds, dim=1, p=2)  # R(a)
    
    # Softmin weights
    weights = F.softmax(-risks / self.temperature, dim=0)  # exp(-R(a)/τ) / Σ exp(-R(a')/τ)
    
    # Weighted average
    E_ref = torch.sum(weights.unsqueeze(1) * E_preds, dim=0)
```

### **3. Alignment Regret (Equation 7)**
**Formula:** `AR_t = ||E_{t+1} - E_{t+1}^ref||_2^2 + κ Σ_j∈N(i) ||E_{j,t+1}||_2`

**Location:** `esaiv3/loss.py`, lines 46-67
```python
def forward(self, E_current: torch.Tensor, E_preds: torch.Tensor,
            E_neighbors: torch.Tensor = None):
    # Compute reference
    E_ref = self.compute_softmin_reference(E_preds)
    
    # Primary regret: ||E - E_ref||^2
    regret = torch.sum((E_current - E_ref) ** 2)  # ||E - E_ref||^2
    
    # Neighbor harm penalty

```

```python
def cosine_similarity(x: torch.Tensor, y: torch.Tensor, eps: float = 1e-8):
    x_norm = torch.norm(x, p=2) + eps
    y_norm = torch.norm(y, p=2) + eps
    similarity = torch.dot(x, y) / (x_norm * y_norm)  # cos(x,y) = x·y / (||x|| ||y||)
    return torch.clamp(similarity, 0.0, 1.0)  # max(0, cos)
```

```python
def _esi_from_hist(self):
    """Compute ESI from sliding window of E norms."""
    vals = list(self._e_hist)
    mu = sum(vals) / len(vals)
    var = sum((x - mu) ** 2 for x in vals) / (len(vals) - 1)
    cv = math.sqrt(var) / (mu + self._eps)  # CV = σ/μ
    return 1.0 / (1.0 + cv)  # ESI = 1/(1 + CV)
```

```python
def anneal_parameter(step: int, init_val: float, final_val: float, total_steps: int):
    """
    Exponential annealing of a parameter.
    """
    if step >= total_steps:
        return final_val
    
    alpha = step / total_steps
    return init_val * np.exp(alpha * np.log(final_val / init_val))  # Exponential decay
```

```python
def compute_laplacian(adjacency: torch.Tensor, normalized: bool = True):
    degree = torch.sum(adjacency, dim=1)
    
    if normalized:
```

### 2️⃣ **Softmin Counterfactual Reference** (Equations 7-8)

**Formulas:**

```python
def compute_softmin_reference(self, E_preds: torch.Tensor):
    """
    Compute softmin reference embedding.
    
    Args:
        E_preds: Predicted IAEs for each action (num_actions, iae_dim)
    
    Returns:
        Reference embedding (iae_dim,)
    """
    # Compute risk: R(a) = ||Ê_{t+1}^(a)||_2
    risks = torch.norm(E_preds, dim=1, p=2)
    
    # Softmin weights: π_ref(a) ∝ exp(-R(a)/τ)
    weights = F.softmax(-risks / self.temperature, dim=0)
    
    # Weighted average: Ê_ref = Σ π_ref(a) Ê^(a)
    E_ref = torch.sum(weights.unsqueeze(1) * E_preds, dim=0)
    
    return E_ref
```

```python
    S = self.compute_similarity_matrix()
    
    # A_weighted = A ⊙ S (element-wise product)
    A = self.base_adjacency * S
    
    # Row-normalize
    row_sums = torch.sum(A, dim=1, keepdim=True) + 1e-8
    A = A / row_sums
    
    return A, S
```

### 1️⃣1️⃣ **Generalized Advantage Estimation (GAE)**

**Formula:**
```
δ_t = r_t + γ V(s_{t+1}) - V(s_t)
A_t = δ_t + γλ δ_{t+1} + (γλ)² δ_{t+2} + ...
```

```python
def compute_gae(rewards, values, dones, gamma=0.99, lambda_=0.95):
    """Compute Generalized Advantage Estimation."""
    advantages = []
    gae = 0
    
    for t in reversed(range(len(rewards))):
        if t == len(rewards) - 1:
            next_value = 0
        else:
            next_value = values[t + 1]
        
        # TD error: δ_t = r_t + γ V_{t+1} - V_t
        delta = rewards[t] + gamma * next_value * (1 - dones[t]) - values[t]
        
        # GAE recursion: A_t = δ_t + γλ A_{t+1}
        gae = delta + gamma * lambda_ * (1 - dones[t]) * gae
        advantages.insert(0, gae)
    
    advantages = torch.tensor(advantages)
    returns = advantages + torch.tensor(values)
    
    return advantages, returns
```

### 1️⃣2️⃣ **PPO-Clip Loss**

**Formula:**
```
L_π = E[min(ρ_t A_t, clip(ρ_t, 1-ε, 1+ε) A_t)]
ρ_t = π_θ(a_t|s_t) / π_{θ_old}(a_t|s_t)
```

**Implementation:** `esaiv3/loss.py` lines 56-72

```python
def forward(self, log_probs, old_log_probs, advantages, values, returns):
    """Compute PPO loss."""
    # Importance ratio: ρ_t = π/π_old
    ratio = torch.exp(log_probs - old_log_probs)
    
    # Clipped surrogate
    surr1 = ratio * advantages
    surr2 = torch.clamp(ratio, 
                        1.0 - self.clip_epsilon, 
                        1.0 + self.clip_epsilon) * advantages
    
    # L_π = -E[min(surr1, surr2)]
    policy_loss = -torch.min(surr1, surr2).mean()
    
    # Value loss (MSE)
    value_loss = F.mse_loss(values, returns)
    
    # Entropy bonus
    entropy = -log_probs.mean()
    
    return policy_loss, value_loss, entropy
```

```python
def cosine_similarity(x: torch.Tensor, y: torch.Tensor, eps: float = 1e-8):
    """
    Compute cosine similarity.
    
    Returns:
        Cosine similarity in [0, 1]
    """
    x_norm = torch.norm(x, p=2) + eps
    y_norm = torch.norm(y, p=2) + eps
    
    # cos(x,y) = x·y / (||x|| ||y||)
    similarity = torch.dot(x, y) / (x_norm * y_norm)
    
    # Clamp to [0, 1] (using max(0, cos) as in paper Eq. 14)
    return torch.clamp(similarity, 0.0, 1.0)
```
